# /etc/postfix/main.cf
# Postfix configuration: test mail sender with OpenDKIM (Ubuntu 22.04)
# Replace younus.com with your own domain if it is different.

# ---------- Basic identity ----------
myhostname = mail.younus.com
mydomain = younus.com
myorigin = $mydomain
smtpd_banner = $myhostname ESMTP
biff = no
append_dot_mydomain = no
compatibility_level = 3.6

# ---------- Network ----------
inet_interfaces = all
inet_protocols = ipv4
mynetworks = 127.0.0.0/8 [::ffff:127.0.0.0]/104 [::1]/128
mydestination = $myhostname, localhost.$mydomain, localhost, $mydomain

# ---------- Mailbox ----------
home_mailbox = Maildir/
mailbox_size_limit = 0
message_size_limit = 10240000
recipient_delimiter = +
alias_maps = hash:/etc/aliases
alias_database = hash:/etc/aliases

# ---------- Relay protection (open relay nahi banna chahiye) ----------
smtpd_relay_restrictions = permit_mynetworks, permit_sasl_authenticated, reject_unauth_destination
smtpd_recipient_restrictions = permit_mynetworks, reject_unauth_destination

# ---------- TLS (default self-signed cert; certbot ke baad neeche wali lines use karo) ----------
smtpd_tls_cert_file = /etc/ssl/certs/ssl-cert-snakeoil.pem
smtpd_tls_key_file = /etc/ssl/private/ssl-cert-snakeoil.key
#smtpd_tls_cert_file = /etc/letsencrypt/live/mail.younus.com/fullchain.pem
#smtpd_tls_key_file = /etc/letsencrypt/live/mail.younus.com/privkey.pem
smtpd_tls_security_level = may
smtp_tls_security_level = may
smtp_tls_CAfile = /etc/ssl/certs/ca-certificates.crt
smtpd_tls_session_cache_database = btree:${data_directory}/smtpd_scache
smtp_tls_session_cache_database = btree:${data_directory}/smtp_scache

# ---------- OpenDKIM (milter) ----------
milter_default_action = accept
milter_protocol = 6
smtpd_milters = inet:localhost:8891
non_smtpd_milters = inet:localhost:8891
